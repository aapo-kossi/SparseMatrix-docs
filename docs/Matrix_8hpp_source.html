290 friend std::vector<T> operator*(const Matrix<T, O, I>& mat,
291 const std::vector<T>& vec)
292 {
293 assert(mat.cols() == vec.size() && "Matrix-vector size mismatch");
294
295 std::vector<T> result(mat.rows(), T{});
296 constexpr bool is_colmajor = O == COL;
297
298 if (mat.is_compressed()) {
299 for (index_t outer = 0; outer < mat.M_outer_ptr.size() - 1; ++outer) {
300 size_t start = mat.M_outer_ptr[outer];
301 size_t end = mat.M_outer_ptr[outer + 1];
302
303 for (size_t k = start; k < end; ++k) {
304 index_t inner = mat.M_inner_indices[k];
305 T val = mat.M_values[k];
306
307 index_t i = is_colmajor ? inner : outer;
308 index_t j = is_colmajor ? outer : inner;
309 result[i] += val * vec[j]; // y[i] += A[i][j] * x[j]
310 }
311 }
312 } else {
313 for (const auto& [key, val] : mat.M_value_map) {
314 index_t row = is_colmajor ? key[1] : key[0];
315 index_t col = is_colmajor ? key[0] : key[1];
316 result[row] += val * vec[col]; // y[i] += A[i][j] * x[j]
317 }
318 }
319
320 return result;
321 }

429 template<typename Self>
430 static decltype(auto) access(Self& self, index_t i, index_t j)
431 {
432 static T zero{};
433 if (self.M_compressed) {
434 constexpr bool is_row_major = O == ROW;
435 index_t outer = is_row_major ? i : j;
436 index_t inner = is_row_major ? j : i;
437
438 size_t start = self.M_outer_ptr[outer];
439 size_t end = self.M_outer_ptr[outer + 1];
440
441 std::span<const index_t> inner_range(self.M_inner_indices.begin() + start,
442 self.M_inner_indices.begin() + end);
443
444 // Do a binary search for outer_val in outer_indices
445 index_t idx = static_cast<index_t>(
446 std::lower_bound(inner_range.begin(), inner_range.end(), inner) -
447 inner_range.begin());
448 if (idx != inner_range.size() && inner_range[idx] == inner) {
449 if constexpr (std::is_const_v<Self>) {
450 T value = self.M_values[idx]; // to consistently rbv when const
451 return value;
452 } else {
453 return self.M_values[idx];
454 }
455 } else {
456 // if non-zero entry is not found and the matrix is compressed, but we
457 // just want to "get a value", return zero
458 // Otherwise abort since elements cannot be inserted while compressed.
459 // Note: with -DNDEBUG trying to do this will return a reference to NaN
460 if constexpr (std::is_const_v<Self>) {
461 return zero;
462 } else {
463 assert(false &&
464 "tried to insert element to matrix in compressed format");
465 static T unreachable{ std::numeric_limits<T>::quiet_NaN() };
466 static T& unreachable_ref = unreachable;
467 return unreachable_ref;
468 }
469 }
470 } else {
471 if constexpr (std::is_const_v<Self>) {
472 // uncompressed const: cannot use operator[] since it's non-const
473 if (auto it = self.M_value_map.find({ i, j });
474 it != self.M_value_map.end()) {
475 T value = it->second;
476 return value;
477 } else {
478 return zero;
479 };
480 } else {
481 // Uncompressed non-const: allow inserting and accessing new values
482 // freely, if no entry exists, new entry is created with default value
483 return self.M_value_map[{ i, j }];
484 }
485 }
486 }

705template<MatrixTraits::Numeric T, MatrixTraits::StorageOrder O, std::integral I>
706Matrix<T, O, I>&
707Matrix<T, O, I>::resize(index_t new_rows, index_t new_cols)
708{
709 // A compressed matrix cannot be resized in a simple way
710 if (is_compressed()) {
711 uncompress();
712 }
713
714 // Delete out-of-bounds elements using C++20 erase_if
715 if (new_rows < M_rows || new_cols < M_cols) {
716 std::erase_if(M_value_map, [&](const auto& pair) {
717 return pair.first[0] >= new_rows || pair.first[1] >= new_cols;
718 });
719 }
720
721 M_rows = new_rows;
722 M_cols = new_cols;
723
724 return *this;
725}

727template<MatrixTraits::Numeric T, MatrixTraits::StorageOrder O, std::integral I>
728template<MatrixTraits::execution::Mode Mode>
729Matrix<T, O, I>&
730Matrix<T, O, I>::compress()
731{
732 if (M_compressed) {
733 throw std::logic_error(
734 "Matrix already compressed. Decompress before recompressing.");
735 }
736
737 if (M_value_map.empty()) {
738 M_compressed = true;
739 return *this;
740 }
741
742 constexpr bool is_row_major = O == ROW;
743 const index_t outer_size = is_row_major ? M_rows : M_cols;
744 const size_t n_entries = M_value_map.size();
745
746 M_values.clear();
747 M_inner_indices.clear();
748 M_outer_ptr.assign(outer_size + 1, 0);
749 M_values.resize(n_entries);
750 M_inner_indices.resize(n_entries);
751
752 const int n_threads = omp_get_max_threads();
753 bool use_parallel = false;
754
755 if constexpr (Mode == MatrixTraits::execution::Mode::Parallel) {
756 use_parallel = true;
757 } else if constexpr (Mode == MatrixTraits::execution::Mode::Sequential) {
758 use_parallel = false;
759 } else {
760 use_parallel = (n_entries >= 10000) && (n_threads > 1);
761 }
762
763 if (use_parallel) {
764 // Creating 'entries' is required for thread-safe random-access iteration
765 // during compression. Unfortunately the largest bottleneck in the parallel
766 // compression, which often makes the parallel version slower even though
767 // the other parts are significantly improved.
768 std::vector<std::pair<std::array<index_t, 2>, T>> entries;
769 entries = { std::make_move_iterator(M_value_map.begin()),
770 std::make_move_iterator(M_value_map.end()) };
771
772 std::vector<index_t> local_counts(n_threads * (outer_size + 1), 0);
773
774 // Every thread 'tid' operates exclusively on its own disjoint segment of
775 // the shared flat array `local_counts`, defined by the range:
776 // [tid * (outer_size + 1), (tid + 1) * (outer_size + 1))
777
778#pragma omp parallel
779 {
780 int tid = omp_get_thread_num();
781 index_t* counts = &local_counts[tid * (outer_size + 1)];
782
783#pragma omp for schedule(static)
784 for (size_t i = 0; i < n_entries; ++i) {
785 const auto& [key, _] = entries[i];
786 index_t outer = is_row_major ? key[0] : key[1];
787 counts[outer + 1]++;
788
789 }
790 }
791
792#pragma omp parallel for schedule(static)
793 for (index_t i = 0; i <= outer_size; ++i) {
794 for (int t = 0; t < n_threads; ++t) {
795 M_outer_ptr[i] += local_counts[t * (outer_size + 1) + i];
796 }
797 }
798
799 for (index_t i = 1; i <= outer_size; ++i) {
800 M_outer_ptr[i] += M_outer_ptr[i - 1];
801 }
802
803 std::vector<std::atomic<size_t>> counters(outer_size + 1);
804
805 for (index_t i = 0; i <= outer_size; ++i) {
806 counters[i].store(M_outer_ptr[i], std::memory_order_relaxed);
807 }
808
809#pragma omp parallel for schedule(static)
810 for (size_t i = 0; i < n_entries; ++i) {
811 const auto& [key, val] = entries[i];
812 const index_t outer = is_row_major ? key[0] : key[1];
813 const index_t inner = is_row_major ? key[1] : key[0];
814
815 size_t pos = counters[outer].fetch_add(1, std::memory_order_relaxed);
816 M_values[pos] = val;
817 M_inner_indices[pos] = inner;
818 }
819 } else {
820 // A std::map cannot be iterated in parallel and we only need to iterate
821 // once
822 size_t n = 0;
823 for (auto it = M_value_map.cbegin(); it != M_value_map.cend();
824 it = M_value_map.erase(it), ++n) {
825 const auto& k = it->first;
826 index_t outer = is_row_major ? k[0] : k[1];
827 index_t inner = is_row_major ? k[1] : k[0];
828 M_values[n] = it->second;
829 M_inner_indices[n] = inner;
830 ++M_outer_ptr[outer + 1];
831 }
832
833 // Prefix sum, doesn't parallelize well
834 for (index_t i = 1; i <= outer_size; ++i) {
835 M_outer_ptr[i] += M_outer_ptr[i - 1];
836 }
837
838 // The map has been emptied already
839 // M_value_map.clear();
840 }
841
842 M_value_map.clear();
843 M_compressed = true;
844 return *this;
845}

847template<MatrixTraits::Numeric T, MatrixTraits::StorageOrder O, std::integral I>
848Matrix<T, O, I>&
849Matrix<T, O, I>::uncompress()
850{
851 if (!M_compressed) {
852 std::cout << "Matrix already uncompressed. Compress before uncompressing."
853 << std::endl;
854 return *this;
855 }
856 constexpr bool is_row_major = O == ROW;
857
858 // This should never have any values since the Matrix is compressed
859 assert(M_value_map.empty());
860 auto it = M_value_map.begin();
861
862 for (index_t outer = 0; outer < M_outer_ptr.size() - 1; ++outer) {
863 size_t outer_start = M_outer_ptr[outer];
864 size_t outer_end = M_outer_ptr[outer + 1];
865
866 for (size_t j = outer_start; j < outer_end; j++) {
867 index_t inner = M_inner_indices[j];
868 T val = M_values[j];
869 std::array<index_t, 2> key =
870 is_row_major ? std::array{ outer, inner } : std::array{ inner, outer };
871 it = M_value_map.emplace_hint(it, key, val);
872 }
873 }
874
875 M_values.clear();
876 M_outer_ptr.clear();
877 M_inner_indices.clear();
878 M_compressed = false;
879 return *this;
880}

884template<typename V, typename W, typename I>
885static void
886scatter_abs_add(std::vector<W>& dst,
887 std::span<const I> idx,
888 std::span<const V> src)
889{
890 assert(idx.size() == src.size());
891
892 for (I i = 0; i < idx.size(); ++i) {
893 if constexpr (std::is_signed_v<V>) {
894 dst[idx[i]] += std::abs(src[i]);
895 } else {
896 dst[idx[i]] += src[i];
897 }
898 }

901// scatter add that tries to vectorize for arithmetic types
902template<typename V, typename I>
903static void
904scatter_abs_add(std::vector<V>& dst,
905 std::span<const I> idx,
906 std::span<const V> src)
907{
908 assert(idx.size() == src.size());
909
910 constexpr I reg_bits = 512;
911 constexpr I value_bits = sizeof(V) * 8;
912 constexpr I block_size = reg_bits / value_bits;
913 I b = 0;
914
915#ifdef __AVX512VL__
916
917 constexpr bool sizes_match = (sizeof(V) == sizeof(I));
918 constexpr bool size_fits = value_bits == 32 || value_bits == 64;
919 constexpr bool vectorizable = sizes_match && size_fits;
920 const index_t blocks = static_cast<index_t>(idx.size() / block_size);
921
922 if constexpr (vectorizable) {
923
924 using op_set = intrinsics<V>;
925
926 for (; b < blocks; ++b) {
927 auto mm_idx = op_set::load_idx(idx.data() + b * 8);
928 auto mm_src = op_set::load(src.data() + b * block_size);
929 mm_src = op_set::abs(mm_src);
930
931 auto mm_dst = op_set::gather(mm_idx, dst.data());
932 mm_dst = op_set::add(mm_src, mm_dst);
933 op_set::scatter(dst.data(), mm_idx, mm_dst);
934 }
935 }
936
937#endif
938
939 // remaining elements one-by-one
940 for (I i = b * block_size; i < idx.size(); ++i) {
941 if constexpr (std::is_signed_v<V>) {
942 dst[idx[i]] += std::abs(src[i]);
943 } else {
944 dst[idx[i]] += src[i];
945 }
946 }
947}

949// scatter add that tries to vectorize for complex types
950template<typename V, typename W, typename I>
951static void
952scatter_abs_add(std::vector<W>& dst,
953 std::span<const I> idx,
954 std::span<const std::complex<V>> src)
955{
956 assert(idx.size() == src.size());
957
958 constexpr I reg_bits = 512;
959 constexpr I value_bits = sizeof(V) * 8;
960 constexpr I block_size = reg_bits / value_bits;
961 I b = 0;
962
963#ifdef __AVX512VL__
964
965 constexpr bool sizes_match = (sizeof(V) == sizeof(I));
966 constexpr bool size_fits = value_bits == 32 || value_bits == 64;
967 constexpr bool vectorizable = sizes_match && size_fits;
968 const I blocks = idx.size() / block_size;
969
970 if constexpr (vectorizable) {
971
972 static_assert(sizeof(V) == sizeof(I));
973 using op_set = intrinsics<V>;
974 using reg = typename op_set::value_reg;
975
976 for (; b < blocks; ++b) {
977
978 // load indices and send prefetch to the data
979 auto mm_idx = op_set::load_idx(idx.data() + b * 8);
980
981#ifdef __AVX512PF__
982 op_set::pf_gather<2>(mm_idx, dst.data());
983#endif
984 // we will shuffle the real and imaginary components
985 // so before scattering we need to make sure the final
986 // ordering is back to how we started
987 __m512i reordering = op_set::get_separated_idx();
988
989 // src takes twice the memory space so we need to load
990 // two blocks for each block of indices
991 auto mm_src1 = op_set::load(src.data() + b * block_size * 2);
992 auto mm_src2 = op_set::load(src.data() + b * block_size * 2 + block_size);
993
994 // separate real and imag components to their own registers
995 reg real, imag;
996 std::tie(real, imag) = op_set::separate_complex(mm_src1, mm_src2);
997
998 // calculate magnitude
999 reg real_squared = op_set::mul(real, real);
1000 reg imag_squared = op_set::mul(imag, imag);
1001 reg norm = op_set::add(real_squared, imag_squared);
1002 reg mag = op_set::sqrt(norm);
1003 mag = op_set::permute(mag, reordering);
1004
1005 // scatter add
1006 auto mm_dst = op_set::gather(mm_idx, dst.data());
1007 mm_dst = op_set::add(mag, mm_dst);
1008 op_set::scatter(dst.data(), mm_idx, mm_dst);
1009 }
1010 }
1011
1012#endif
1013
1014 // remaining elements one-by-one
1015 for (I i = b * block_size; i < idx.size(); ++i) {
1016 dst[idx[i]] += std::abs(src[i]);
1017 }
1018}

1025template<MatrixTraits::Norm N, MatrixTraits::execution::Mode M>
1026auto
1027algebra::Matrix<T, O, I>::_norm_compressed() const
1028{
1029
1030 if constexpr (N == MatrixTraits::Norm::Frobenius) {
1031 const auto norm_func = [](T x) {
1032 if constexpr (std::is_unsigned_v<T>) {
1033 return x;
1034 } else {
1035 return std::norm(x);
1036 }
1037 };
1038 const auto reduce_norm = [this, norm_func](const auto& policy) {
1039 const auto sqnorm = std::transform_reduce(policy,
1040 M_values.cbegin(),
1041 M_values.cend(),
1042 norm_t{},
1043 std::plus<norm_t>{},
1044 norm_func);
1045 return std::sqrt(sqnorm);
1046 };
1047 return MatrixTraits::execution::maybe_par_unseq<M>(reduce_norm,
1048 M_values.size());
1049
1050 } else {
1051
1052 constexpr bool is_colmajor = O == COL;
1053
1054 constexpr bool is_onenorm = N == MatrixTraits::Norm::One;
1055
1056 // Easily parallelisable case where sums are calculated
1057 // across contiguous blocks of data
1058 if constexpr (is_onenorm == is_colmajor) {
1059 auto constexpr abs_fun = [](auto x) {
1060 if constexpr (std::is_unsigned_v<T>) {
1061 return x;
1062 } else {
1063 return std::abs(x);
1064 }
1065 };
1066
1067 assert(M_outer_ptr.size() > 0);
1068 auto idx =
1069 std::views::iota(0u, M_outer_ptr.size() - 1) | std::views::common;
1070
1071 auto inner_sums = [this](index_t outer) {
1072 std::span<T const> inner_range = (*this)[outer];
1073
1074 norm_t inner_norm = std::transform_reduce(std::execution::unseq,
1075 inner_range.begin(),
1076 inner_range.end(),
1077 norm_t{},
1078 std::plus{},
1079 abs_fun);
1080 return inner_norm;
1081 };
1082 auto reduce_norm = [inner_sums, idx](const auto& policy) {
1083 auto max_fn = [](norm_t a, norm_t b) { return std::max(a, b); };
1084 return std::transform_reduce(
1085 policy, idx.begin(), idx.end(), norm_t{}, max_fn, inner_sums);
1086 };
1087 return MatrixTraits::execution::maybe_par<M>(reduce_norm, idx.size());
1088
1089 } else {
1090 // Otherwise we need to perform scatter add
1091 // still can vectorize manually IF AVX512 is available
1092 // but that is painful
1093 const index_t n_sums = is_onenorm ? M_cols : M_rows;
1094 ThreadedAccumulator<base_t>::reset();
1095 ThreadedAccumulator<base_t>::set_size(n_sums);
1096 auto inner_sum = [this](index_t outer) {
1097 auto inner_iter = M_inner_indices.begin();
1098 std::span<index_t const> inner_indices{
1099 inner_iter + M_outer_ptr[outer], inner_iter + M_outer_ptr[outer + 1]
1100 };
1101 std::span<T const> inner_values = (*this)[outer];
1102
1103 detail::scatter_abs_add(
1104 ThreadedAccumulator<base_t>::vec(), inner_indices, inner_values);
1105 };
1106
1107 assert(M_outer_ptr.size() > 0);
1108 auto accumulate = [this, inner_sum](const auto& policy) {
1109 auto idx =
1110 std::views::iota(0u, M_outer_ptr.size() - 1) | std::views::common;
1111 std::for_each(policy, idx.begin(), idx.end(), inner_sum);
1112 return ThreadedAccumulator<base_t>::result();
1113 };
1114 std::vector<norm_t> total_sums = MatrixTraits::execution::maybe_par<M>(
1115 accumulate, M_outer_ptr.size() - 1);
1116
1117 auto reduce_max = [&total_sums](const auto& policy) {
1118 auto max_op = [](norm_t a, norm_t b) { return std::max(a, b); };
1119 return std::reduce(
1120 policy, total_sums.begin(), total_sums.end(), norm_t{}, max_op);
1121 };
1122 return MatrixTraits::execution::maybe_par_unseq<M>(reduce_max,
1123 total_sums.size());
1124 }
1125 }
1126}

1142template<MatrixTraits::Norm N, MatrixTraits::execution::Mode M, class It>
1143auto
1144algebra::Matrix<T, O, I>::_norm_inner(const It& iterable,
1145 const auto& get_value) const
1146{
1147 if constexpr (N == MatrixTraits::Norm::Frobenius) {
1148 auto calculate_norm = [&get_value, &iterable](const auto& policy) {
1149 auto norm_func = [&get_value](const auto& x) {
1150 return std::norm(std::get<2>(get_value(x)));
1151 };
1152 norm_t sqnorm = std::transform_reduce(policy,
1153 iterable.begin(),
1154 iterable.end(),
1155 norm_t{ 0 },
1156 std::plus<norm_t>{},
1157 norm_func);
1158 return std::sqrt(sqnorm);
1159 };
1160 return MatrixTraits::execution::maybe_par_unseq<M>(calculate_norm,
1161 iterable.size());
1162 } else {
1163 constexpr bool is_onenorm = N == MatrixTraits::Norm::One;
1164 index_t n_sums = is_onenorm ? M_cols : M_rows;
1165
1166 std::vector<norm_t> sums(n_sums, norm_t{});
1167
1168 auto accumulate = [s = sums.data(), &get_value](const auto& idx) {
1169 index_t col, row;
1170 T v;
1171 std::tie(row, col, v) = get_value(idx);
1172
1173 norm_t norm;
1174 if constexpr (std::is_signed_v<norm_t>) {
1175 norm = std::abs(v);
1176 } else {
1177 norm = v;
1178 }
1179 if constexpr (is_onenorm) {
1180 s[col] += norm;
1181 } else {
1182 s[row] += norm;
1183 }
1184 };
1185
1186 std::for_each(
1187 std::execution::seq, iterable.begin(), iterable.end(), accumulate);
1188
1189 auto reduce_max = [&sums](const auto& policy) {
1190 auto max_fun = [](const norm_t& a, const norm_t& b) -> norm_t {
1191 return a > b ? a : b;
1192 };
1193 return std::reduce(
1194 policy, sums.begin(), sums.end(), norm_t{ 0 }, max_fun);
1195 };
1196 return MatrixTraits::execution::maybe_par_unseq<M>(reduce_max, sums.size());
1197 }
